    <script>
        // الـ API Key تبعك من الصورة
        const apiKey = 'AIzaSyBw8v3Rda0kw8M0WpOnuoy6XvL1Z92EgcI'; 
        const systemInstruction = "You are an expert Clinical Decision Assistant for Pharmacists in Jordan. Your goal is to guide the pharmacist through clinical guidelines (e.g., Hypertension treatment) using a step-by-step 'Decision Tree' approach. Rules: 1. Ask ONLY ONE question at a time. 2. WAIT for the user's response. 3. Suggest Jordanian commercial brands at the end. Reply in Arabic.";

        let chatHistory = [];

        async function startChat() {
            const inputField = document.getElementById('user-input');
            const message = inputField.value.trim();
            if (!message) return;

            addMessage('user', message);
            inputField.value = '';

            const loadingId = 'loading-' + Date.now();
            addMessage('ai', 'جاري التفكير...', loadingId);

            chatHistory.push({ "role": "user", "parts": [{ "text": message }] });

            try {
                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        system_instruction: { parts: [{ text: systemInstruction }] },
                        contents: chatHistory
                    })
                });

                const data = await response.json();
                document.getElementById(loadingId).remove();

                if (data.candidates && data.candidates[0]) {
                    const aiReply = data.candidates[0].content.parts[0].text;
                    addMessage('ai', aiReply);
                    chatHistory.push({ "role": "model", "parts": [{ "text": aiReply }] });
                } else {
                    addMessage('ai', "حدث خطأ في استلام الرد. تأكد من إعدادات الـ API.");
                }
            } catch (error) {
                if(document.getElementById(loadingId)) document.getElementById(loadingId).remove();
                addMessage('ai', "خطأ في الاتصال. يرجى المحاولة مرة أخرى.");
                console.error(error);
            }
        }

        function addMessage(sender, text, id = null) {
            const chatBox = document.getElementById('chat-box');
            const div = document.createElement('div');
            div.className = `message ${sender}-message`;
            if (id) div.id = id;
            div.innerHTML = text.replace(/\n/g, '<br>').replace(/\*\*(.*?)\*\*/g, '<b>$1</b>');
            chatBox.appendChild(div);
            chatBox.scrollTop = chatBox.scrollHeight;
        }

        document.getElementById('send-btn').onclick = startChat;
        document.getElementById('user-input').onkeypress = (e) => { if (e.key === 'Enter') startChat(); };
    </script>
